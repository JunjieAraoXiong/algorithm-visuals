完整 Python 代码

按你给出的代码整理缩进；上方动画用 capacity = 3 演示。

class DoublyLinkedListNode:
    """Store a cache entry and links to its neighboring nodes."""

    def __init__(self, key: int, val: int) -> None:
        self.key = key
        self.val = val
        self.next = self.prev = None


class LRUCache:
    """A fixed-capacity cache that evicts the least recently used entry."""

    def __init__(self, capacity: int) -> None:
        """Create a cache with the given capacity."""
        self.capacity = capacity

        # Map each key directly to its linked-list node.
        self.hashmap = {}

        # Dummy nodes mark the two ends of the list.
        # Nodes near tail were used more recently.
        self.head = DoublyLinkedListNode(-1, -1)
        self.tail = DoublyLinkedListNode(-1, -1)
        self.head.next = self.tail
        self.tail.prev = self.head

    def add_to_tail(self, node: DoublyLinkedListNode) -> None:
        """Place node just before tail, marking it most recently used."""
        prev_node = self.tail.prev

        # First, point the node to its new neighbors.
        node.prev = prev_node
        node.next = self.tail

        # Replace the old prev_node <-> tail connection
        # with prev_node <-> node <-> tail.
        prev_node.next = node
        self.tail.prev = node

    def remove_node(self, node: DoublyLinkedListNode) -> None:
        """Unlink node from the list without changing the hashmap."""
        # Change previous <-> node <-> next into previous <-> next.
        node.prev.next = node.next
        node.next.prev = node.prev

    def get(self, key: int) -> int:
        """Return a key's value, or -1 if absent; mark it recently used."""
        if key not in self.hashmap:
            return -1

        node = self.hashmap[key]

        # A successful get counts as using this entry.
        self.remove_node(node)
        self.add_to_tail(node)

        return node.val

    def put(self, key: int, value: int) -> None:
        """Add or update a key, evicting the least recent if necessary."""
        if key in self.hashmap:
            # Update the existing node and move it to the recent end.
            node = self.hashmap[key]
            node.val = value
            self.remove_node(node)
            self.add_to_tail(node)
            return

        if len(self.hashmap) == self.capacity:
            # The real node immediately after head is least recent.
            least_recent = self.head.next
            self.remove_node(least_recent)
            del self.hashmap[least_recent.key]

        # A new entry is the most recently used.
        node = DoublyLinkedListNode(key, value)
        self.hashmap[key] = node
        self.add_to_tail(node)

为什么这两个结构要一起用？

hashmap[key] 直接给出节点。链表从 head（最久未使用）排到 tail（最近使用）；head 和 tail 是不存业务键值的哨兵。

  1. get(key)：查表。没找到就返回 -1，链表不变；找到了就把同一个节点移到 tail 前，再返回它的值。
  2. put(key, value)：已有键时，修改原节点的 val 并将它移到尾部，节点身份不变。新键且已满时，先摘下并删掉最旧节点的表项，然后创建新节点、建立映射并接到尾部。
  3. remove_node 只改左右邻居的 .next / .prev。摘下的节点暂时还保留旧的 prev / next，但从 head 已走不到它。若同一节点重新插入，add_to_tail 会覆盖这两个字段；若被淘汰，局部变量 least_recent 会暂时引用它，函数结束后若无其他引用便可回收。

上面的「逐行看双向链表的指针」可以对照每一次字段赋值；「完整 Python 代码」保留了整个类的执行顺序。

查找和局部改线都是 O(1)，所以 get、put 平均 O(1)；最多保存 capacity 个有效节点，空间 O(capacity)。get 未命中不会改变最近使用顺序。
